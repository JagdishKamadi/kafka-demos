## If face any issue while starting kakfa, delete the tmp folder from C:drive

# Start ZooKeeper
=> .\bin\windows\zookeeper-server-start.bat config\zookeeper.properties

# Start Kafka
=> .\bin\windows\kafka-server-start.bat config\server.properties

# To Start multiple Kafka server
=> .\bin\windows\kafka-server-start.bat config\server-2.properties

# Create topic
=> .\bin\windows\kafka-topics.bat --create --topic <topic-name> --bootstrap-server localhost:9092

# To Send the message via CLI
=> .\bin\windows\kafka-console-producer.bat --topic payment-events --bootstrap-server localhost:9092 --property "parse.key=true" --property "key.separator=:"

# List topics
=> .\bin\windows\kafka-topics.bat --list --bootstrap-server localhost:9092

# Describe topic
=> .\bin\windows\kafka-topics.bat --describe --topic <topic-name> --bootstrap-server localhost:9092

# Delete topic
=> .\bin\windows\kafka-topics.bat --delete --topic <topic-name> --bootstrap-server localhost:9092

# Start producer
=> .\bin\windows\kafka-console-producer.bat --topic <topic-name> --bootstrap-server localhost:9092

# Start consumer
=> .\bin\windows\kafka-console-consumer.bat --topic <topic-name> --bootstrap-server localhost:9092

# Start consumer with printing key
=> .\bin\windows\kafka-console-consumer.bat --topic product-created-events-topic --bootstrap-server localhost:9092 --property print.key=true

# Consumer from beginning
=> .\bin\windows\kafka-console-consumer.bat --topic <topic-name> --from-beginning --bootstrap-server localhost:9092

# Consumer with group
=> .\bin\windows\kafka-console-consumer.bat --topic <topic-name> --group <group-name> --bootstrap-server localhost:9092

# List consumer groups
=> .\bin\windows\kafka-consumer-groups.bat --list --bootstrap-server localhost:9092

# Describe consumer group
=> .\bin\windows\kafka-consumer-groups.bat --describe --group <group-name> --bootstrap-server localhost:9092
